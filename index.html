package org.example.strings; public class StringTransformation { public static String solve(String s) { StringBuilder result = new StringBuilder(); int i = 0; while (i < s.length()) { char current = s.charAt(i); boolean isVowel = "aeiouAEIOU".indexOf(current) != -1; if (isVowel) { result.append('*'); i++; } else if (Character.isLetter(current) && i + 1 < s.length() && current == s.charAt(i + 1)) { result.append("10"); i += 2; } else { result.append(current); i++; } } return result.toString(); } public static void main(String[] args) { String input = "aabbec"; System.out.println(solve(input)); // **10*c } }
